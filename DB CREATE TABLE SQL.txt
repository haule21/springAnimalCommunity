create table BOARD(
	CATEGORY_NUM INT UNSIGNED not null,
	BOARD_NUM INT UNSIGNED not_null,
	REGISTERED_DATE DATETIME not null default CURRENT_TIMESTAMP,
	TITLE VARCHAR(300) not null,
	WRITER VARCHAR(33) not null,
	MODIFIED_DATE DATETIME not null default CURRENT_TIMESTAMP,
	DELETED ENUM('Y','N') not null default 'N',
	EXIST_IMGFILE not null,
	CONTENT BLOB(65535) not null,
	constraint BOARD_PK primary key (CATEGORY_NUM, BOARD_NUM, REGISTERED_DATE)
) engine=INNODB default CHARSET=UTF8MB4
partition by range( CATEGORY_NUM )(
	PARTITION C1 VALUES LESS THAN (200) ENGINE = INNODB
	PARTITION C2 VALUES LESS THAN (300) ENGINE = INNODB
	PARTITION C3 VALUES LESS THAN (400) ENGINE = INNODB
	PARTITION C4 VALUES LESS THAN (500) ENGINE = INNODB
	PARTITION C5 VALUES LESS THAN (600) ENGINE= INNODB
	PARTITION C6 VALUES LESS THAN (700) ENGINE = INNODB
	PARTITION C7 VALUES LESS THAN (800) ENGINE = INNODB
	PARTITION C8 VALUES LESS THAN (900) ENGINE = INNODB
	PARTITION C9 VALUES LESS THAN (1000) ENGINE = INNODB
	PARTITION CM VALUES LESS THAN (MAXVALUE) ENGINE = INNODB
)

create table USERS(
	USERID VARCHAR(16) not null,
	PASSWORD VARCHAR(88) not null,
	NICKNAME VARCHAR(33) not null,
	REGISTERED_DATE DATETIME not null default CURRENT_TIMESTAMP,
	ROLE_NUM tinyint(1) not null,
	DELETED ENUM('Y','N') not null default 'N',
	constraint USERS_PK primary key(USERID),
	constraint USERS_FK foreign key(ROLE_NUM) references USER_ROLE(ROLE_NUM)
) engine=INNODB default CHARSET=UTF8MB4

create table USER_PRIVACY(
	USERID VARCHAR(16) not null,
	NAME VARCHAR(15) not null,
	EMAIL VARCHAR(100) not null,
	PHONE_NUMBER VARCHAR(15) not null,
	ADDRESS VARCHAR(300),
	constraint USER_PRIVACY_PK primary key(USERID),
	constraint USER_PRIVACY_FK foreign key(USERID) references USERS(USERID)
) engine=INNODB default CHARSET=UTF8MB4

create table USER_STAT(

) engine=INNODB default CHARSET=UTF8MB4

create table USER_ICON_HISTORY(

	MADE_USERID VARCHAR(16) not null,
	constraint USER_STAT_PK primary key(USERID, ICON_NUM)
) engine=INNODB default CHARSET=UTF8MB4

create procedure USER_INSERT_PROCEDURE(
	USERID_ VARCHAR(16), PASSWORD_ VARCHAR(88), NICKNAME_ VARCHAR(33),
	NAME_ VARCHAR(15), EMAIL_ VARCHAR(100), PHONE_NUMBER_ VARCHAR(15), ADDRESS_ VARCHAR(300))
BEGIN
	START TRANSACTION;
	insert into USERS values(
		USERID_, PASSWORD_, NICKNAME_, DEFAULT, 1, DEFAULT
	);
	insert into USER_PRIVACY values(
		USERID_, NAME_, EMAIL_, PHONE_NUMBER_, ADDRESS_
	);
	COMMIT;
END
